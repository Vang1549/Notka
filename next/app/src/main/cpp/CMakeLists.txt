cmake_minimum_required(VERSION 3.22)
project(notka_audio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(notka_dsp STATIC sampler.cpp)
set_target_properties(notka_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  find_package(oboe REQUIRED CONFIG)
  add_library(notka_audio SHARED bridge.cpp)
  target_link_libraries(notka_audio PRIVATE notka_dsp oboe::oboe android log)
  target_link_options(notka_audio PRIVATE "-Wl,-z,max-page-size=16384")
else()
  enable_testing()
  add_executable(dsp_test ../../test/cpp/dsp_test.cpp)
  target_link_libraries(dsp_test PRIVATE notka_dsp)
  target_include_directories(dsp_test PRIVATE .)
  add_test(NAME sampler COMMAND dsp_test)
endif()
